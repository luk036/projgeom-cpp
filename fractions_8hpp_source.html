740 [[nodiscard]] constexpr auto rationalize(const std::array<Frac, 3>& values)
741 -> std::array<std::int64_t, 3> {
742 std::int64_t denom = 1;
743 for (const auto& v : values) {
744 denom = lcm(denom, v.den());
745 }
746 std::array<std::int64_t, 3> ints{};
747 for (std::size_t i = 0; i < values.size(); ++i) {
748 ints[i] = values[i].num() * (denom / values[i].den());
749 }
750 std::int64_t divisor = 0;

752 divisor = gcd(divisor, v);
