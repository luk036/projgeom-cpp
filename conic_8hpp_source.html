28 [[nodiscard]] constexpr auto isqrt(std::int64_t n) -> std::int64_t {
29 if (n < 0) {
30 return -1;
31 }
32 if (n < 2) {
33 return n;
34 }
35 std::int64_t lo = 1;
36 std::int64_t hi = n / 2 + 1;
37 while (lo <= hi) {
38 const std::int64_t mid = lo + (hi - lo) / 2;
39 const std::int64_t sq = mid * mid;
40 if (sq == n) {
41 return mid;
42 }
43 if (sq < n) {
44 lo = mid + 1;
45 } else {
46 hi = mid - 1;
47 }
48 }
49 return hi;
50 }

100 [[nodiscard]] constexpr auto rational_sqrt(const Frac& value) -> std::optional<Frac> {
101 if (value < Frac{0, 1}) {
102 return std::nullopt;
103 }
104 const auto n = isqrt(value.num());
105 const auto d = isqrt(value.den());
106 if (n * n == value.num() && d * d == value.den()) {
107 return Frac{n, d};
108 }
109 return std::nullopt;
110 }

244 if (det3(matrix_) == Frac{0, 1}) {
245 throw std::domain_error{"Cannot compute the pole of a degenerate conic"};
246 }
247 const auto adj = adjugate3(matrix_);
248 const auto& l = line.coord;

269 [[nodiscard]] auto intersect(const PgLine& line) const -> std::vector<PgPoint> {
270 const Vec3 line_coord{
271 {Frac{line.coord[0], 1}, Frac{line.coord[1], 1}, Frac{line.coord[2], 1}}};
272
273 std::vector<Vec3> vectors;
274 for (std::size_t j = 0; j < 3; ++j) {
275 Vec3 axis{{Frac{0, 1}, Frac{0, 1}, Frac{0, 1}}};
276 axis[j] = Frac{1, 1};
277 const auto candidate = detail::cross3(line_coord, axis);
278 if (!detail::is_zero(candidate)) {
279 vectors.push_back(candidate);
280 }
281 }
282 const Vec3 origin = vectors[0];
283 Vec3 direction{};
284 for (std::size_t i = 1; i < vectors.size(); ++i) {
285 if (!detail::is_zero(detail::cross3(origin, vectors[i]))) {
286 direction = vectors[i];
287 break;
288 }
289 }
290
291 const auto quad = detail::quadratic_form(matrix_, direction);
292 const auto lin = Frac{2, 1} * detail::bilinear_form(matrix_, origin, direction);
293 const auto constant = detail::quadratic_form(matrix_, origin);
294
295 const auto point_at = [&](const Frac& t) -> PgPoint {
296 return PgPoint{
297 rationalize(Vec3{{origin[0] + t * direction[0], origin[1] + t * direction[1],
298 origin[2] + t * direction[2]}})};
299 };
300
301 if (quad == Frac{0, 1}) {
302 if (lin == Frac{0, 1}) {
303 throw std::domain_error{"line lies on the conic"};
304 }
305 return {point_at(-constant / lin)};
306 }
307
308 const auto disc = lin * lin - Frac{4, 1} * quad * constant;
309 if (disc < Frac{0, 1}) {
310 return {};
311 }
312 const auto root = detail::rational_sqrt(disc);
313 if (!root.has_value()) {
314 throw std::domain_error{"intersection requires a quadratic surd"};
315 }
316 if (root.value() == Frac{0, 1}) {
317 return {point_at(-lin / (Frac{2, 1} * quad))};
318 }
319 return {point_at((-lin + root.value()) / (Frac{2, 1} * quad)),
320 point_at((-lin - root.value()) / (Frac{2, 1} * quad))};
321 }
