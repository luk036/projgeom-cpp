62 [[nodiscard]] constexpr auto inverse3(const Mat3& m) -> Mat3 {
63 const auto det = det3(m);
64 if (det == Frac{0, 1}) {
65 throw std::domain_error{"Cannot invert singular matrix"};
66 }
67 const auto inv_det = Frac{1, 1} / det;
68 const auto adj = adjugate3(m);
69 Mat3 result{};
70 for (std::size_t i = 0; i < 3; ++i) {

72 result[i][j] = adj[i][j] * inv_det;
