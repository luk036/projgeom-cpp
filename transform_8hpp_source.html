144 constexpr auto compose(const Transform& other) const -> Transform {
145 Mat3x3 result{};

206 constexpr auto inverse() const -> Transform {
207 const auto& m = matrix_;
208 const auto& a = m[0][0];
209 const auto& b = m[0][1];
210 const auto& c = m[0][2];
211 const auto& d = m[1][0];
212 const auto& e = m[1][1];
213 const auto& f = m[1][2];
214 const auto& g = m[2][0];
215 const auto& h = m[2][1];
216 const auto& i_ = m[2][2];
217
218 const auto det = a * (e * i_ - f * h) - b * (d * i_ - f * g) + c * (d * h - e * g);
219 if (det == Frac{0, 1}) {
220 throw std::domain_error{"Cannot invert singular transformation matrix"};
221 }
222 const auto inv_det = Frac{1, 1} / det;
223
224 Mat3x3 result{};
225 result[0][0] = inv_det * (e * i_ - f * h);
226 result[0][1] = inv_det * (c * h - b * i_);
227 result[0][2] = inv_det * (b * f - c * e);
228 result[1][0] = inv_det * (f * g - d * i_);
229 result[1][1] = inv_det * (a * i_ - c * g);
230 result[1][2] = inv_det * (c * d - a * f);
231 result[2][0] = inv_det * (d * h - e * g);
232 result[2][1] = inv_det * (b * g - a * h);
233 result[2][2] = inv_det * (a * e - b * d);
234 return Transform{result};
235 }

301 [[nodiscard]] inline auto projective_transform(const std::array<PgPoint, 4>& src,
302 const std::array<PgPoint, 4>& dst) -> Transform {
303 const auto support = [](const std::array<PgPoint, 4>& pts) -> Mat3 {
304 Mat3 m{};
305 for (std::size_t i = 0; i < 3; ++i) {
306 for (std::size_t j = 0; j < 3; ++j) {
307 m[i][j] = Frac{pts[j].coord[i], 1};
308 }
309 }
310 return m;
311 };
312 const auto as_vec = [](const PgPoint& pt) -> Vec3 {
313 return Vec3{{Frac{pt.coord[0], 1}, Frac{pt.coord[1], 1}, Frac{pt.coord[2], 1}}};
314 };
315
316 const auto src_inverse = inverse3(support(src));
317 const auto dst_inverse = inverse3(support(dst));
318 const auto alpha = matvec3(src_inverse, as_vec(src[3]));
319 const auto beta = matvec3(dst_inverse, as_vec(dst[3]));
320
321 Mat3 diagonal{};
322 for (std::size_t i = 0; i < 3; ++i) {
323 if (alpha[i] == Frac{0, 1} || beta[i] == Frac{0, 1}) {
324 throw std::domain_error{"correspondences are not in general position"};
325 }
326 diagonal[i][i] = beta[i] / alpha[i];
327 }
328 return Transform{matmul3(matmul3(support(dst), diagonal), src_inverse)};
329 }
